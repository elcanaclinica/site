// =====================================================================
// REGRAS DE SEGURANÇA DO FIREBASE STORAGE — Clínica Elcana
// =====================================================================
// Cole estas regras em:
// Firebase Console → Storage → Rules → Editar → Publicar
// =====================================================================

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Pasta de imagens do site
    match /site-images/{fileName} {

      // LEITURA: qualquer pessoa pode ver as imagens (necessário para o site)
      allow read: if true;

      // UPLOAD: somente admin autenticado
      allow write: if request.auth != null

        // Tamanho máximo: 5MB
        && request.resource.size <= 5 * 1024 * 1024

        // Somente imagens (JPG, PNG, WEBP)
        && request.resource.contentType.matches('image/(jpeg|png|webp)')

        // Nome do arquivo: apenas caracteres seguros
        && fileName.matches('[a-zA-Z0-9_\\-\\.]+');

      // DELETE: somente admin autenticado
      allow delete: if request.auth != null;
    }

    // TUDO o mais está BLOQUEADO
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
