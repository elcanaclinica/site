// =====================================================================
// REGRAS DE SEGURANÇA DO FIRESTORE — Clínica Elcana
// =====================================================================
// Cole estas regras em:
// Firebase Console → Firestore Database → Rules → Editar → Publicar
// =====================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Documento único de configuração do site
    match /config/site {

      // LEITURA: qualquer pessoa pode ler (o site público precisa disso)
      allow read: if true;

      // ESCRITA: SOMENTE usuários autenticados pelo Firebase Auth
      // (ou seja, somente quem fez login com e-mail/senha válidos)
      allow write: if request.auth != null

        // Validação de tamanho: impede documentos gigantes
        && request.resource.data.keys().size() <= 10

        // Validação de tipos de campo
        && request.resource.data.texts is map
        && request.resource.data.contact is map
        && request.resource.data.wallpaper is map
        && request.resource.data.procedures is list

        // Limite de procedimentos
        && request.resource.data.procedures.size() <= 30

        // Campos de texto não podem ser absurdamente grandes
        && request.resource.data.texts.heroTitle.size()     <= 120
        && request.resource.data.texts.heroSub.size()       <= 400
        && request.resource.data.texts.aboutTitle.size()    <= 120
        && request.resource.data.texts.aboutText.size()     <= 1000
        && request.resource.data.texts.contactPhrase.size() <= 400
        && request.resource.data.contact.address.size()     <= 200
        && request.resource.data.contact.phone.size()       <= 15
        && request.resource.data.contact.instagram.size()   <= 50;
    }

    // TUDO o mais está BLOQUEADO por padrão (nenhum outro acesso)
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
